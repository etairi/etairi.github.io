{{- /* GitHub Pages cannot redirect on the server, so every entry in data/redirects.yaml becomes a small HTML page at the old path that forwards to the new address. */ -}} {{- range $from, $to := (partialCached "data.html" .).redirects }} {{- $target := $to | absURL }} {{- $html := printf `Moved

This page has moved to %[1]s.

` $target }} {{- $path := printf "%s/index.html" (trim $from "/") }} {{- (resources.FromString $path $html).Publish }} {{- end }}
