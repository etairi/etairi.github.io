(dict "key" $key)) }} {{- end }} {{- $items = $items | append (merge $t (dict "occasions" (sort $occasions "key" "desc") "latest" $latest "year" (time.AsTime $latest).Year)) }} {{- end }} {{- $items = sort $items "latest" "desc" }}
